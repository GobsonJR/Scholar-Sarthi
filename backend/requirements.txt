# --- LayoutLMv3 document understanding (Phase 5) ---
# torch is intentionally NOT pinned in this file: a plain `pip install -r
# requirements.txt` resolves it from PyPI's default index, which would pull
# torch's default (CUDA-bundled, multi-GB) build. Install the CPU build
# FIRST, from PyTorch's own CPU index, then install this file:
#   pip install torch==2.14.0 --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
#
# transformers is pinned to 4.44.2, NOT the newer 5.x line Phase 5 originally
# used. Reason: IndicTrans2's custom remote-code model
# (ai4bharat/indictrans2-indic-en-dist-200M) was written against
# transformers' ~4.4x-era internal API (tokenizer attribute handling,
# `transformers.onnx` still present) and its `trust_remote_code=True` files
# are not maintained for transformers 5.x — loading it under 5.17.0 raised a
# real AttributeError deep in tokenizer init, not a simple import path issue.
# LayoutLMv3 (Phase 5) is a mainline, long-stable architecture and was
# re-verified working correctly on 4.44.2 after this downgrade — see the
# Phase 6 section of README.md for the regression check.
transformers==4.44.2
sentencepiece==0.2.2
# IndicTransToolkit has a Cython extension with NO prebuilt Windows wheel —
# it builds from source and needs a C++ compiler (MSVC Build Tools on
# Windows; gcc/clang are typically already present on Linux/macOS).
